	int max=0;
	for(int i=0;i<=9;i++) {
		int sum=0;
		for(int j =0;j<n;j++) {
			int cha = i;
			int chb = a[j];
			String cval = Integer.toString(chb);
			String copy = cval.replaceAll(String.valueOf(cha), "");
			int lval; 
			if(copy.isEmpty())
